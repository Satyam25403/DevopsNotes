we will create an ec2 instance:
terraform{
  required_providers{
    aws ={
      source = "hashicorp/aws"
      version = "-> 4.0"
    }
  }
}

provider "aws" {
  region = "ap-south-1"
  access_key = <access_key>
  secret_key = <secret_key>
}
resource "aws_s3_bucket" <some_name>{
  bucket = <bucket_name>
  acl = "private"
  tags = {
    "name" : "value"
  }
}
resource "aws_instance" <instance_name>{
    ami = "ami-xyz"
    instance_type: "t2.micro"
    tags ={
        "num" =var.num_var                  =>take from variables.tf
    }
    key_name = <an_IAM_User>            => a user having proper ec2 creation/full access permission
}


in outputs.tf file:(used for accessing created resources) after the .tf file is applied
output "hosted_zone_id" {
    value = aws_s3_bucket.<bucket_name>.hosted_zone_id
}
output "ip"{
    value = aws_instance.<instancen_name>.public_ip
}

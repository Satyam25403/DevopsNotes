## Production Load Balancer with SSL

### Complete Production Setup

**Scenario:**
- 3 Node.js apps on ports 3001, 3002, 3003
- Each has `APP_NAME` environment variable (app1, app2, app3)
- Nginx as HTTPS entry point and load balancer
- SSL termination at Nginx

**nginx.conf:**
```nginx
# Main context
user www-data;
worker_processes auto;  # Auto-detect CPU cores
pid /run/nginx.pid;

events {
    worker_connections 1024;
    use epoll;  # Efficient on Linux
}

http {
    # MIME types
    include /etc/nginx/mime.types;
    default_type application/octet-stream;

    # Logging
    access_log /var/log/nginx/access.log;
    error_log /var/log/nginx/error.log;

    # Performance
    sendfile on;
    tcp_nopush on;
    tcp_nodelay on;
    keepalive_timeout 65;

    # Upstream servers
    upstream nodejscluster {
        least_conn;  # Least busy algorithm
        server 127.0.0.1:3001 max_fails=3 fail_timeout=30s;
        server 127.0.0.1:3002 max_fails=3 fail_timeout=30s;
        server 127.0.0.1:3003 max_fails=3 fail_timeout=30s;
    }

    # HTTP to HTTPS redirect
    server {
        listen 80;
        server_name example.com www.example.com;

        location / {
            return 301 https://$server_name$request_uri;
        }
    }

    # HTTPS server
    server {
        listen 443 ssl http2;
        server_name example.com www.example.com;

        # SSL certificates
        ssl_certificate /etc/nginx/ssl/nginx-selfsigned.crt;
        ssl_certificate_key /etc/nginx/ssl/nginx-selfsigned.key;

        # SSL settings
        ssl_protocols TLSv1.2 TLSv1.3;
        ssl_prefer_server_ciphers on;
        ssl_ciphers 'ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256';
        ssl_session_cache shared:SSL:10m;
        ssl_session_timeout 10m;

        # Security headers
        add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;
        add_header X-Frame-Options "SAMEORIGIN" always;
        add_header X-Content-Type-Options "nosniff" always;
        add_header X-XSS-Protection "1; mode=block" always;

        # Proxy to Node.js cluster
        location / {
            proxy_pass http://nodejscluster;
            proxy_http_version 1.1;

            # Essential headers
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            proxy_set_header X-Forwarded-Host $host;
            proxy_set_header X-Forwarded-Port $server_port;

            # WebSocket support
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";

            # Timeouts
            proxy_connect_timeout 60s;
            proxy_send_timeout 60s;
            proxy_read_timeout 60s;
        }
    }
}
```

### Docker Compose Setup

```yaml
version: '3.8'

services:
  app1:
    image: node:18-alpine
    environment:
      - APP_NAME=app1
      - PORT=3000
    ports:
      - "3001:3000"
    volumes:
      - ./app:/app
    command: node /app/server.js

  app2:
    image: node:18-alpine
    environment:
      - APP_NAME=app2
      - PORT=3000
    ports:
      - "3002:3000"
    volumes:
      - ./app:/app
    command: node /app/server.js

  app3:
    image: node:18-alpine
    environment:
      - APP_NAME=app3
      - PORT=3000
    ports:
      - "3003:3000"
    volumes:
      - ./app:/app
    command: node /app/server.js

  nginx:
    image: nginx:alpine
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./nginx.conf:/etc/nginx/nginx.conf:ro
      - ./ssl:/etc/nginx/ssl:ro
    depends_on:
      - app1
      - app2
      - app3
```

**Node.js server.js:**
```javascript
const express = require('express');
const app = express();
const port = process.env.PORT || 3000;
const appName = process.env.APP_NAME || 'unknown';

app.get('/', (req, res) => {
    const ip = req.headers['x-forwarded-for'] || req.connection.remoteAddress;
    const proto = req.headers['x-forwarded-proto'];
    
    res.send(`
        Request served by ${appName}
        Client IP: ${ip}
        Protocol: ${proto}
    `);
});

app.listen(port, () => {
    console.log(`${appName} listening on port ${port}`);
});
```

**Run:**
```bash
# Start all services
docker-compose up -d

# Test HTTPS
curl -k https://localhost
# Output rotates: app1, app2, app3
```

## Security Hardening

### Essential Security Headers

```nginx
server {
    listen 443 ssl http2;
    server_name example.com;

    # Hide Nginx version
    server_tokens off;

    # Security headers
    add_header Strict-Transport-Security "max-age=31536000; includeSubDomains; preload" always;
    add_header X-Frame-Options "DENY" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-XSS-Protection "1; mode=block" always;
    add_header Referrer-Policy "no-referrer-when-downgrade" always;
    add_header Content-Security-Policy "default-src 'self' https:; script-src 'self' 'unsafe-inline'; style-src 'self' 'unsafe-inline'" always;

    location / {
        # Your configuration
    }
}
```

### Blocking Bad Bots

```nginx
# Block bad user agents
map $http_user_agent $bad_bot {
    default 0;
    ~*bot 1;
    ~*crawler 1;
    ~*spider 1;
    ~*scraper 1;
}

server {
    if ($bad_bot) {
        return 403;
    }
}
```

### IP Whitelisting

```nginx
# Allow only specific IPs
location /admin {
    allow 192.168.1.0/24;
    allow 10.0.0.0/8;
    deny all;
    
    proxy_pass http://backend;
}
```

### Protecting Against DDoS

```nginx
# Limit connections per IP
limit_conn_zone $binary_remote_addr zone=conn_limit:10m;

server {
    # Max 10 connections per IP
    limit_conn conn_limit 10;
}
```

## Performance Optimization

### Gzip Compression

```nginx
http {
    # Enable gzip
    gzip on;
    gzip_vary on;
    gzip_proxied any;
    gzip_comp_level 6;
    gzip_types text/plain text/css text/xml text/javascript 
               application/json application/javascript application/xml+rss 
               application/rss+xml application/atom+xml image/svg+xml 
               text/x-component text/x-cross-domain-policy;
    gzip_disable "MSIE [1-6]\.";
}
```

### Browser Caching

```nginx
# Cache static assets
location ~* \.(jpg|jpeg|png|gif|ico|css|js|svg|woff|woff2|ttf|eot)$ {
    expires 1y;
    add_header Cache-Control "public, immutable";
    access_log off;
}

# Cache HTML with short expiry
location ~* \.html$ {
    expires 1h;
    add_header Cache-Control "public, must-revalidate";
}
```

### Connection Optimization

```nginx
http {
    # Keepalive connections
    keepalive_timeout 65;
    keepalive_requests 100;

    # Client body
    client_body_buffer_size 128k;
    client_max_body_size 10m;
    client_body_timeout 12;
    client_header_timeout 12;

    # Sendfile
    sendfile on;
    tcp_nopush on;
    tcp_nodelay on;

    # Buffer sizes
    client_header_buffer_size 1k;
    large_client_header_buffers 4 8k;
}
```

### Upstream Keepalive

```nginx
upstream backend {
    server 127.0.0.1:3000;
    server 127.0.0.1:3001;
    
    # Keep connections open
    keepalive 32;
}

server {
    location / {
        proxy_pass http://backend;
        proxy_http_version 1.1;
        proxy_set_header Connection "";
    }
}
```

### Custom Log Formats

```nginx
http {
    # Detailed logging
    log_format detailed '$remote_addr - $remote_user [$time_local] '
                       '"$request" $status $body_bytes_sent '
                       '"$http_referer" "$http_user_agent" '
                       'rt=$request_time uct="$upstream_connect_time" '
```

## Rate Limiting

### Basic Rate Limiting

```nginx
http {
    # Define rate limit zone
    limit_req_zone $binary_remote_addr zone=api_limit:10m rate=10r/s;

    server {
        location /api/ {
            # Apply rate limit
            limit_req zone=api_limit burst=20 nodelay;
            proxy_pass http://backend;
        }
    }
}
```

**Explanation:**
- `rate=10r/s` - 10 requests per second
- `burst=20` - Allow bursts up to 20 requests
- `nodelay` - Don't delay excess requests

### Different Limits for Different Paths

```nginx
http {
    # API rate limit
    limit_req_zone $binary_remote_addr zone=api:10m rate=10r/s;
    
    # Login rate limit (stricter)
    limit_req_zone $binary_remote_addr zone=login:10m rate=5r/m;
    
    # Static files (generous)
    limit_req_zone $binary_remote_addr zone=static:10m rate=50r/s;

    server {
        location /api/ {
            limit_req zone=api burst=20;
            proxy_pass http://backend;
        }

        location /login {
            limit_req zone=login burst=5;
            proxy_pass http://backend;
        }

        location /static/ {
            limit_req zone=static burst=100;
            root /var/www;
        }
    }
}
```

### Custom Error Page

```nginx
server {
    location /api/ {
        limit_req zone=api_limit burst=20 nodelay;
        limit_req_status 429;
        
        proxy_pass http://backend;
    }

    error_page 429 /rate_limit.html;
    location = /rate_limit.html {
        internal;
        root /var/www/errors;
    }
}
```

## Geographic Restrictions

### GeoIP Module

```bash
# Install GeoIP
sudo apt install libnginx-mod-http-geoip

# Download GeoIP database
wget https://github.com/maxmind/geoipupdate/releases/download/...
```

**Configuration:**
```nginx
http {
    geoip_country /usr/share/GeoIP/GeoIP.dat;

    # Block specific countries
    map $geoip_country_code $allowed_country {
        default yes;
        CN no;  # China
        RU no;  # Russia
    }

    server {
        if ($allowed_country = no) {
            return 403;
        }

        location / {
            # Your config
        }
    }
}
```

### Allow Specific Countries

```nginx
http {
    geoip_country /usr/share/GeoIP/GeoIP.dat;

    map $geoip_country_code $allowed {
        default no;
        US yes;  # United States
        GB yes;  # United Kingdom
        CA yes;  # Canada
    }

    server {
        if ($allowed = no) {
            return 403 "Access denied from your country";
        }
    }
}
```

## API Gateway Configuration

### Complete API Gateway

```nginx
http {
    # Rate limiting
    limit_req_zone $binary_remote_addr zone=api:10m rate=100r/s;

    # Caching
    proxy_cache_path /var/cache/nginx levels=1:2 keys_zone=api_cache:10m;

    # Upstream services
    upstream user_service {
        server 127.0.0.1:3001;
        server 127.0.0.1:3002;
    }

    upstream product_service {
        server 127.0.0.1:4001;
        server 127.0.0.1:4002;
    }

    upstream order_service {
        server 127.0.0.1:5001;
    }

    server {
        listen 443 ssl http2;
        server_name api.example.com;

        # SSL config
        ssl_certificate /etc/nginx/ssl/cert.pem;
        ssl_certificate_key /etc/nginx/ssl/key.pem;

        # Security headers
        add_header X-Frame-Options "DENY" always;
        add_header X-Content-Type-Options "nosniff" always;

        # User API
        location /api/users {
            limit_req zone=api burst=50;
            
            proxy_pass http://user_service;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            
            # Cache GET requests
            proxy_cache api_cache;
            proxy_cache_valid 200 5m;
        }

        # Product API
        location /api/products {
            limit_req zone=api burst=50;
            
            proxy_pass http://product_service;
            proxy_set_header Host $host;
            
            # Cache GET requests
            proxy_cache api_cache;
            proxy_cache_valid 200 10m;
        }

        # Order API (no caching - mutations)
        location /api/orders {
            limit_req zone=api burst=20;
            
            proxy_pass http://order_service;
            proxy_set_header Host $host;
            
            # No caching for orders
            proxy_no_cache 1;
        }

        # Health check endpoint
        location /health {
            access_log off;
            return 200 "OK\n";
            add_header Content-Type text/plain;
        }
    }
}
```

### API Versioning

```nginx
server {
    # v1 API
    location /api/v1/ {
        proxy_pass http://backend_v1/;
    }

    # v2 API
    location /api/v2/ {
        proxy_pass http://backend_v2/;
    }

    # Default to latest
    location /api/ {
        proxy_pass http://backend_v2/;
    }
```

## Microservices Routing

### Service Mesh Pattern

```nginx
http {
    # Service discovery
    upstream auth_service {
        server auth:8080;
    }

    upstream user_service {
        server user:8080;
    }

    upstream payment_service {
        server payment:8080;
    }

    server {
        listen 80;

        # Authentication
        location /auth/ {
            proxy_pass http://auth_service/;
            proxy_set_header Host $host;
        }

        # Users (requires auth)
        location /users/ {
            # Auth subrequest
            auth_request /auth/verify;
            
            proxy_pass http://user_service/;
            proxy_set_header Host $host;
        }

        # Payment (requires auth)
        location /payment/ {
            auth_request /auth/verify;
            
            proxy_pass http://payment_service/;
            proxy_set_header Host $host;
        }

        # Auth verification endpoint
        location = /auth/verify {
            internal;
            proxy_pass http://auth_service/verify;
            proxy_pass_request_body off;
            proxy_set_header Content-Length "";
        }
    }
}
```

## Docker Integration

### Nginx in Docker

**Dockerfile:**
```dockerfile
FROM nginx:alpine

# Copy custom config
COPY nginx.conf /etc/nginx/nginx.conf

# Copy SSL certificates
COPY ssl/ /etc/nginx/ssl/

# Copy static files
COPY html/ /usr/share/nginx/html/

# Health check
HEALTHCHECK --interval=30s --timeout=3s \
  CMD wget --quiet --tries=1 --spider http://localhost/health || exit 1

EXPOSE 80 443

CMD ["nginx", "-g", "daemon off;"]
```

**Build and run:**
```bash
docker build -t my-nginx .
docker run -d -p 80:80 -p 443:443 --name nginx my-nginx
```

## Troubleshooting

### Common Issues

**1. 502 Bad Gateway**

**Check backend:**
```bash
# Backend running?
curl http://localhost:3000

# Nginx can reach backend?
docker exec nginx curl http://backend:3000
```

**Fix:**
```nginx
upstream backend {
    # Add more time
    server backend:3000 max_fails=3 fail_timeout=30s;
}

location / {
    proxy_connect_timeout 60s;
    proxy_read_timeout 60s;
}
```

**2. 413 Request Entity Too Large**

**Fix:**
```nginx
http {
    client_max_body_size 100M;
}
```

**3. WebSocket Connection Failed**

**Fix:**
```nginx
location /ws/ {
    proxy_pass http://backend;
    proxy_http_version 1.1;
    proxy_set_header Upgrade $http_upgrade;
    proxy_set_header Connection "upgrade";
}
```

### Debug Logging

```nginx
error_log /var/log/nginx/error.log debug;

server {
    access_log /var/log/nginx/access.log detailed;
    error_log /var/log/nginx/error.log debug;
}
```

LIFECYCLE IN ONE SCRIPT
docker pull nginx:alpine
# image lives on disk. nothing running yet.
docker run -d --name web nginx:alpine
# now there's a container. image unchanged.
docker stop web
# container gone. image still there. reuse anytime.

ECR lifecycle policy JSON
{
"rules": [
// Rule 1: nuke untagged after 1 day
{ "rulePriority": 1,
"selection": { "tagStatus": "untagged",
"countType": "sinceImagePushed",
"countNumber": 1, "countUnit": "days" },
"action": { "type": "expire" } },
// Rule 2: keep only last 10 tagged
{ "rulePriority": 2,
"selection": { "tagStatus": "tagged",
"tagPrefixList": ["v"],
"countType": "imageCountMoreThan",
"countNumber": 10 },
"action": { "type": "expire" } }]}
APPLY VIA AWS CLI
$ aws ecr put-lifecycle-policy \
--repository-name myapp \
--lifecycle-policy-text file://policy.json
# preview before applying:
$ aws ecr start-lifecycle-policy-preview \
--repository-name myapp \
--lifecycle-policy-text file://policy.json

LOCAL PRUNE CLI
terminal
# remove dangling images only
docker image prune
# remove ALL unused images older than 24h
docker image prune -a \
--filter "until=24h"
# images with a specific label
docker image prune \
--filter "label=env=ci"
# no prompt, for scripts/cron
docker image prune -a -f --filter "until=168h"
--filter OPTIONS
until=24h → older than 24 hours
until=168h → older than 1 week
until=2024-01-01T00:00:00 → before date
label=key → has this label
label=k=v → label equals value
